! R4 - P pur + Route Reflector VPNv4
hostname R4
!
ip cef
!
interface Loopback0
 ip address 50.0.0.4 255.255.255.255
!
interface FastEthernet0/0
 description Coeur vers R2
 ip address 10.2.4.2 255.255.255.252
 mpls ip
 no shutdown
!
interface FastEthernet0/1
 description Coeur vers R3
 ip address 10.3.4.2 255.255.255.252
 mpls ip
 no shutdown
!
mpls label protocol ldp
mpls ldp router-id Loopback0 force
!
router ospf 1
 network 10.2.4.0 0.0.0.3 area 0
 network 10.3.4.0 0.0.0.3 area 0
 network 50.0.0.4 0.0.0.0 area 0
!
router bgp 65000
 neighbor 50.0.0.1 remote-as 65000
 neighbor 50.0.0.1 update-source Loopback0
 neighbor 50.0.0.2 remote-as 65000
 neighbor 50.0.0.2 update-source Loopback0
 neighbor 50.0.0.3 remote-as 65000
 neighbor 50.0.0.3 update-source Loopback0
 address-family vpnv4
  neighbor 50.0.0.1 activate
  neighbor 50.0.0.1 send-community extended
  neighbor 50.0.0.1 route-reflector-client
  neighbor 50.0.0.2 activate
  neighbor 50.0.0.2 send-community extended
  neighbor 50.0.0.2 route-reflector-client
  neighbor 50.0.0.3 activate
  neighbor 50.0.0.3 send-community extended
  neighbor 50.0.0.3 route-reflector-client
 exit-address-family
!
end
